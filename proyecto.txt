import { fetchProducts } from './supabase.js';

const state = {
  products: [],
  loading: false,
  error: null,
};

const $ = (sel, ctx = document) => ctx.querySelector(sel);

function formatCLP(value) {
  return new Intl.NumberFormat('es-CL', {
    style: 'currency',
    currency: 'CLP',
    maximumFractionDigits: 0,
  }).format(value);
}

function escapeHtml(str) {
  return String(str ?? '')
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#39;');
}

function renderGrid() {
  const grid = $('#catalogGrid');
  if (!grid) return;

  if (state.loading) {
    grid.innerHTML = '<p class="catalog-status">Cargando amigurumis… 🧶</p>';
    return;
  }
  if (state.error) {
    grid.innerHTML = `<p class="catalog-status catalog-error">No pudimos cargar el catálogo: ${escapeHtml(state.error)}</p>`;
    return;
  }
  if (!state.products.length) {
    grid.innerHTML = `
      <div class="catalog-status catalog-empty">
        <p>🧶 Aún no hay productos en stock.</p>
        <p class="catalog-empty-hint">
          Si esperabas ver productos, revisa que la tabla <code>products</code> en Supabase
          tenga filas y que las políticas RLS permitan lectura pública.
        </p>
      </div>
    `;
    return;
  }

  grid.innerHTML = state.products
    .map((p) => {
      const stock = Number(p.stock ?? 0);
      const disabled = stock <= 0 ? 'disabled' : '';
      const media = p.image_url
        ? `<img src="${escapeHtml(p.image_url)}" alt="${escapeHtml(p.name)}" loading="lazy" />`
        : `<span class="catalog-card-emoji">${escapeHtml(p.emoji || '🧸')}</span>`;
      const desc = p.desc_text || p.desc || '';
      const ctaLabel = stock > 0 ? 'Añadir al Carrito' : 'Sin stock';
      return `
        <article class="catalog-card">
          <div class="catalog-card-media">${media}</div>
          <div class="catalog-card-body">
            <h3 class="catalog-card-title">${escapeHtml(p.name)}</h3>
            <p class="catalog-card-desc">${escapeHtml(desc)}</p>
            <span class="catalog-card-price">${formatCLP(p.price)}</span>
            <button class="catalog-card-cta" type="button" ${disabled} data-add="${escapeHtml(p.id)}">
              ${ctaLabel}
            </button>
          </div>
        </article>
      `;
    })
    .join('');

  grid.querySelectorAll('[data-add]').forEach((btn) => {
    btn.addEventListener('click', () => {
      if (typeof window.addToCart === 'function') {
        window.addToCart(btn.dataset.add);
      }
    });
  });
}

async function loadCatalog() {
  const grid = $('#catalogGrid');
  if (!grid) return;

  state.loading = true;
  state.error = null;
  renderGrid();

  try {
    state.products = await fetchProducts();
    if (typeof window.registerProducts === 'function') {
      window.registerProducts(state.products);
    }
  } catch (err) {
    const msg = err?.message || 'Error desconocido';
    state.error = /failed to fetch/i.test(msg)
      ? 'No se pudo conectar con el servidor. Verifica tu conexión o que el sitio se esté sirviendo por HTTP (no file://).'
      : msg;
  } finally {
    state.loading = false;
    renderGrid();
  }
}

/* Compatibilidad: si algún trigger antiguo llama a openCatalog, solo recarga */
export async function openCatalog() {
  await loadCatalog();
}

export function closeCatalog() {
  /* No-op: el catálogo ahora es una vista de página, no un modal */
}

window.openCatalog = openCatalog;
window.closeCatalog = closeCatalog;

  const grid = $('#catalogGrid');

  state.loading = true;

const supabaseKey = 'sb_publishable_DrVZWQpgCbF4pZfWh0UdMQ_cLRvIAut';

export const supabase = createClient(supabaseUrl, supabaseKey);

const TABLE = 'productos';

function normalizePayload(payload = {}) {
  const out = {
    name: (payload.name ?? '').toString().trim(),
    desc_text: (payload.desc_text ?? payload.desc ?? '').toString().trim(),
    price: Number(payload.price) || 0,
    stock: Number(payload.stock) || 0,
    emoji: (payload.emoji ?? '🧸').toString().trim() || '🧸',
    image_url: payload.image_url ? payload.image_url.toString().trim() : null,
  };
  if (payload.category != null) out.category = payload.category.toString().trim();
  if (payload.id != null && String(payload.id).trim() !== '') {
    out.id = payload.id;
  }
  return out;
}

export async function fetchProducts() {
  const { data, error } = await supabase
    .from(TABLE)
    .select('*')
    .order('id', { ascending: true });
  if (error) throw error;
  return data ?? [];
}

export async function createProduct(payload) {
  const { data, error } = await supabase
    .from(TABLE)
    .insert(normalizePayload(payload))
    .select()
    .single();
  if (error) throw error;
  return data;
}

export async function updateProduct(id, payload) {
  const clean = normalizePayload(payload);
  delete clean.id;
  const { data, error } = await supabase
    .from(TABLE)
    .update(clean)
    .eq('id', id)
    .select()
    .single();
  if (error) throw error;
  return data;
}

export async function deleteProduct(id) {
  const { error } = await supabase.from(TABLE).delete().eq('id', id);
  if (error) throw error;
}

export async function signInAdmin(email, password) {
  const { data, error } = await supabase.auth.signInWithPassword({ email, password });
  if (error) throw error;
  return data;
}

export async function signOutAdmin() {
  const { error } = await supabase.auth.signOut();
  if (error) throw error;
}

export async function getSession() {
  const { data } = await supabase.auth.getSession();
  return data.session;
}

/* ---------- Datos de productos ---------- */
let PRODUCTS = [];

/* Productos dinámicos (Supabase) registrados desde js/catalog.js */
const dynamicProducts = new Map();

function registerProducts(list) {
  if (!Array.isArray(list)) return;
  list.forEach((p) => {
    if (p && p.id != null) {
      dynamicProducts.set(String(p.id), {
        id: String(p.id),
        name: p.name,
        desc: p.desc || '',
        price: Number(p.price) || 0,
        emoji: p.emoji || '🧸',
        image_url: p.image_url || null,
      });
    }
  });
}

function findProduct(id) {
  const key = String(id);
  return dynamicProducts.get(key) || PRODUCTS.find((p) => String(p.id) === key);
}

window.registerProducts = registerProducts;

/* ---------- Utilidades ---------- */
const formatCLP = (value) => {
  const n = Number(value);
  const safe = Number.isFinite(n) ? n : 0;
  return 'CLP $' + safe.toLocaleString('es-CL', { maximumFractionDigits: 0 });
};

/* ---------- Estado del carrito ---------- */
const STORAGE_KEY = 'patitas-de-lana-cart';

let cart = loadCart();

function loadCart() {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    return raw ? JSON.parse(raw) : [];
  } catch {
    return [];
  }
}

function saveCart() {
  localStorage.setItem(STORAGE_KEY, JSON.stringify(cart));
}

/* ---------- Carga de productos desde Supabase ----------
   El render del catálogo en stock vive en js/catalog.js (módulo ES).
   Aquí solo registramos los productos para que el carrito pueda
   resolverlos por id cuando se hace clic en "Añadir al Carrito". */
async function fetchProducts() {
  try {
    const { fetchProducts: fetchFromSupabase } = await import('./js/supabase.js');
    const data = await fetchFromSupabase();
    if (!Array.isArray(data)) return;

    PRODUCTS = data.map((p) => ({
      id: p.id,
      name: p.name,
      desc: p.desc_text || p.desc || '',
      price: Number(p.price) || 0,
      emoji: p.emoji || '🧸',
      image_url: p.image_url || null,
      stock: p.stock ?? 0,
    }));

    registerProducts(PRODUCTS);
  } catch (err) {
    console.error('Error al cargar productos de Supabase:', err.message);
  }
}

/* ---------- Carrito ---------- */
function addToCart(id) {
  const product = findProduct(id);
  if (!product) return;

  const key = String(product.id);
  const existing = cart.find((item) => String(item.id) === key);
  if (existing) {
    existing.qty += 1;
  } else {
    cart.push({
      id: key,
      name: product.name,
      price: product.price,
      emoji: product.emoji,
      image_url: product.image_url || null,
      qty: 1,
    });
  }

  saveCart();
  renderCart();
  showToast(`✨ ${product.name} añadido al carrito`);
}

window.addToCart = addToCart;

function removeFromCart(id) {
  const key = String(id);
  cart = cart.filter((item) => String(item.id) !== key);
  saveCart();
  renderCart();
}

function changeQty(id, delta) {
  const key = String(id);
  const item = cart.find((i) => String(i.id) === key);
  if (!item) return;
  item.qty += delta;
  if (item.qty <= 0) {
    removeFromCart(id);
    return;
  }
  saveCart();
  renderCart();
}

function getTotal() {
  return cart.reduce((sum, item) => sum + item.price * item.qty, 0);
}

function getCount() {
  return cart.reduce((sum, item) => sum + item.qty, 0);
}

function renderCart() {
  const itemsEl = $('#cartItems');
  const totalEl = $('#cartTotal');
  const countEl = $('#cartCount');
  const checkoutBtn = $('#checkoutBtn');

  if (!itemsEl) return;

  if (cart.length === 0) {
    itemsEl.innerHTML = '<p class="cart-empty">Tu carrito está vacío 🧶<br>¡Agrega un amigurumi!</p>';
  } else {
    itemsEl.innerHTML = cart.map((item) => `
      <div class="cart-item">
        <div class="cart-item-img" aria-hidden="true">${item.emoji}</div>
        <div class="cart-item-info">
          <h4>${item.name}</h4>
          <span>${formatCLP(item.price)} c/u</span>
          <div class="qty-controls">
            <button data-action="dec" data-id="${item.id}" aria-label="Quitar uno">−</button>
            <span>${item.qty}</span>
            <button data-action="inc" data-id="${item.id}" aria-label="Agregar uno">+</button>
          </div>
        </div>
        <button class="cart-item-remove" data-action="remove" data-id="${item.id}" aria-label="Eliminar">🗑️</button>
      </div>
    `).join('');
  }

  if (totalEl) totalEl.textContent = formatCLP(getTotal());
  if (countEl) countEl.textContent = getCount();
  if (checkoutBtn) checkoutBtn.disabled = cart.length === 0;
}

/* Delegación de eventos del carrito */
document.addEventListener('click', (e) => {
  const btn = e.target.closest('[data-action]');
  if (!btn) return;
  const { action, id } = btn.dataset;
  if (action === 'inc') changeQty(id, 1);
  if (action === 'dec') changeQty(id, -1);
  if (action === 'remove') removeFromCart(id);
});

/* ---------- Modal de pago ---------- */
const paymentModal = $('#paymentModal');

function openPayment() {
  if (cart.length === 0) return;
  const amountEl = $('#paymentAmount');
  if (amountEl) amountEl.textContent = formatCLP(getTotal());
  if (paymentModal) {
    paymentModal.removeAttribute('hidden');
    paymentModal.style.display = '';
  }
  if (overlay) overlay.style.display = '';
}

function closePayment() {
  if (paymentModal) {
    paymentModal.style.display = 'none';
    paymentModal.setAttribute('hidden', '');
  }
  if (overlay && (!cartDrawer || !cartDrawer.classList.contains('open'))) {
    overlay.style.display = 'none';
  }
}

/* ---------- Selector de modo ---------- */
const modeCards = $$('.mode-card');
const modeHint = $('#modeHint');
const sections = $$('[data-mode-section]');

function selectMode(mode) {
  modeCards.forEach((card) => {
    const isSelected = card.dataset.mode === mode;
    card.classList.toggle('selected', isSelected);
    card.setAttribute('aria-checked', String(isSelected));
  });

  sections.forEach((sec) => {
    sec.classList.toggle('active', sec.dataset.modeSection === mode);
  });

  if (modeHint) {
    modeHint.textContent = mode === 'stock'
      ? '🛍️ ¡Genial! Explora el catálogo en stock.'
      : '✨ ¡Perfecto! Cuéntanos tu idea personalizada.';
  }

  // "stock" navega a la vista de catálogo; "custom" navega a personalizados
  if (mode === 'stock') {
    setTimeout(() => {
      window.location.href = 'catalogo.html';
    }, 250);
  } else if (mode === 'custom') {
    setTimeout(() => {
      window.location.href = 'personalizados.html';
    }, 250);
  }
}

modeCards.forEach((card) => {
  card.addEventListener('click', () => selectMode(card.dataset.mode));
  card.addEventListener('keydown', (e) => {
    if (e.key === 'Enter' || e.key === ' ') {
      e.preventDefault();
      selectMode(card.dataset.mode);
    }
  });
});

/* ---------- Carga de imágenes de referencia ---------- */
const uploadInput = $('#imagenes');
const uploadBtn = $('#uploadBtn');
const uploadArea = $('#uploadArea');
const uploadPreview = $('#uploadPreview');

let uploadedFiles = [];

function renderUploadPreview() {
  if (!uploadPreview) return;
  uploadPreview.innerHTML = uploadedFiles.map((file, idx) => `
    <div class="upload-thumb">
      <img src="${file.url}" alt="${file.name}" />
      <button type="button" data-remove="${idx}" aria-label="Quitar imagen">✕</button>
    </div>
  `).join('');
}

function addFiles(fileList) {
  const files = Array.from(fileList || []).filter((f) => f.type.startsWith('image/'));
  files.forEach((file) => {
    uploadedFiles.push({
      name: file.name,
      url: URL.createObjectURL(file),
    });
  });
  renderUploadPreview();
}

if (uploadBtn && uploadInput) {
  uploadBtn.addEventListener('click', () => uploadInput.click());
  uploadInput.addEventListener('change', (e) => {
    addFiles(e.target.files);
    uploadInput.value = '';
  });
}

if (uploadArea) {
  ['dragenter', 'dragover'].forEach((evt) => {
    uploadArea.addEventListener(evt, (e) => {
      e.preventDefault();
      uploadArea.classList.add('dragover');
    });
  });
  ['dragleave', 'drop'].forEach((evt) => {
    uploadArea.addEventListener(evt, (e) => {
      e.preventDefault();
      uploadArea.classList.remove('dragover');
    });
  });
  uploadArea.addEventListener('drop', (e) => {
    if (e.dataTransfer?.files) addFiles(e.dataTransfer.files);
  });
}

if (uploadPreview) {
  uploadPreview.addEventListener('click', (e) => {
    const btn = e.target.closest('[data-remove]');
    if (!btn) return;
    const idx = Number(btn.dataset.remove);
    const removed = uploadedFiles.splice(idx, 1)[0];
    if (removed?.url) URL.revokeObjectURL(removed.url);
    renderUploadPreview();
  });
}

if (customForm) {
  customForm.addEventListener('submit', (e) => {
    e.preventDefault();

    const tipo = $('#tipo').value;
    const tamano = $('#tamano').value;
    const colores = $('#colores').value.trim() || 'A definir';
    const nombre = $('#nombre').value.trim();
    const detalles = $('#detalles').value.trim() || 'Sin detalles adicionales';

    if (!nombre) {
      showToast('Por favor, ingresa tu nombre 😊');
      $('#nombre').focus();

    }

    const imagenesTxt = uploadedFiles.length
      ? uploadedFiles.map((f) => `  - ${f.name}`).join('\n')
      : '  (sin imágenes adjuntas)';

    let mensaje =
      `¡Hola Patitas de Lana! 🧶 Quiero cotizar un amigurumi personalizado:\n\n` +
      `• Tipo: ${tipo}\n` +
      `• Tamaño: ${tamano}\n` +
      `• Colores: ${colores}\n` +
      `• Detalles: ${detalles}\n`;

    if (idea) mensaje += `• Idea: ${idea}\n`;

    mensaje +=
      `• Imágenes de referencia:\n${imagenesTxt}\n\n` +
      `Mi nombre es ${nombre}. ¡Gracias!`;

    const url = `https://wa.me/${WHATSAPP_NUMBER}?text=${encodeURIComponent(mensaje)}`;
    window.open(url, '_blank', 'noopener');
  });
}

/* ---------- Toast ---------- */
let toastTimer;
function showToast(message) {
  const toast = $('#toast');
  if (!toast) return;
  toast.textContent = message;
  toast.classList.add('show');
  clearTimeout(toastTimer);
  toastTimer = setTimeout(() => toast.classList.remove('show'), 2800);
}
